find_package(CutefishToolkit REQUIRED)
add_executable(makos-polkit-agent main.cpp polkitagentlistener.cpp authsession.cpp dialog.cpp)
set_target_properties(makos-polkit-agent PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON AUTOMOC OFF)
target_link_libraries(makos-polkit-agent PRIVATE Cutefish::Toolkit)
add_executable(makos-auth-protocol-test test_protocol.cpp)
set_target_properties(makos-auth-protocol-test PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON AUTOMOC OFF)
add_executable(makos-auth-session-test test_session.cpp authsession.cpp)
set_target_properties(makos-auth-session-test PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON AUTOMOC OFF)
target_link_libraries(makos-auth-session-test PRIVATE Cutefish::ToolkitBase)

endforeach()
